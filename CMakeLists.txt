cmake_minimum_required(VERSION 3.20)
project(FAULTLINE VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(faultline_core src/core/rng/deterministic_rng.cpp)
target_include_directories(faultline_core PUBLIC src)
target_compile_features(faultline_core PUBLIC cxx_std_20)
function(faultline_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
  endif()
endfunction()
faultline_warnings(faultline_core)

    src/world/generation/relationship_generation.cpp src/world/generation/actor_hash.cpp src/world/generation/actor_validation.cpp)
target_link_libraries(faultline_world PUBLIC faultline_core)
faultline_warnings(faultline_world)
